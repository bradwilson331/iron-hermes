### New `word_wrapped_line_count` Function

Full implementation (char-level, mirrors ratatui WordWrapper { trim: false }):

```rust
pub(crate) fn word_wrapped_line_count(line: &str, width: usize) -> usize {
    if line.is_empty() || width == 0 {
        return 1;
    }

    let mut rows: usize = 1;
    let mut current_row_width: usize = 0;
    let mut pending_word_width: usize = 0;
    let mut pending_ws_width: usize = 0;
    let mut in_word = false;

    let flush_word = |rows: &mut usize,
                      current_row_width: &mut usize,
                      pending_ws_width: &mut usize,
                      pending_word_width: &mut usize,
                      width: usize| {
        if *pending_word_width == 0 { return; }
        let needed = *current_row_width + *pending_ws_width + *pending_word_width;
        if needed <= width {
            *current_row_width = needed;
        } else {
            *rows += 1;
            *current_row_width = 0;
            *pending_ws_width = 0;
            if *pending_word_width <= width {
                *current_row_width = *pending_word_width;
            } else {
                let word_rows = (*pending_word_width + width - 1) / width;
                *rows += word_rows - 1;
                let remainder = *pending_word_width % width;
                *current_row_width = if remainder == 0 { width } else { remainder };
            }
        }
        *pending_word_width = 0;
        *pending_ws_width = 0;
    };

    for c in line.chars() {
        let char_w = UnicodeWidthChar::width(c).unwrap_or(0);
        if c.is_whitespace() {
            if in_word {
                flush_word(&mut rows, &mut current_row_width, &mut pending_ws_width,
                           &mut pending_word_width, width);
                in_word = false;
            }
            if char_w > 0 {
                if current_row_width + pending_ws_width + char_w > width {
                    rows += 1;
                    current_row_width = 0;
                    pending_ws_width = char_w;
                } else {
                    pending_ws_width += char_w;
                }
            }
        } else {
            in_word = true;
            pending_word_width += char_w;
        }
    }
    if pending_word_width > 0 {
        flush_word(&mut rows, &mut current_row_width, &mut pending_ws_width,
                   &mut pending_word_width, width);
    }
    rows
}
```

## Algorithm Divergences from RESEARCH §3

The RESEARCH §3 provided a `split_whitespace`-based sketch followed by a char-level "Recommended mitigation". This implementation follows the **char-level** version as mandated by:
- PLAN `<action>` step 2: "char-level word/whitespace iterator"
- RESEARCH §6 Pitfall 3: `split_whitespace` silently drops leading spaces

One structural difference from the sketch: the implementation uses a closure `flush_word` to avoid code duplication between the mid-line flush (when whitespace follows a word) and the end-of-line flush. This is functionally equivalent to the algorithm described.
