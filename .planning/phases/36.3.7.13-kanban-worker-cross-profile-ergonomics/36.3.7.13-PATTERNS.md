```rust
/// Phase 36.3.7.13 D-B2: remove all tools NOT in `allowed`.
///
/// Used by `filter_for_goal_mode` to drop disallowed tools from the
```

```rust
/// Phase 36.3.7.13 D-B2: after register_default_tools + register_kanban_tools,
/// drop disallowed tools from the registry when in goal-mode worker mode.
///
/// Only fires when BOTH conditions hold:
///   - HERMES_KANBAN_TASK is set (worker mode)
///   - HERMES_KANBAN_GOAL_MODE=1 is set (goal mode)
///
/// The preset name is read from HERMES_KANBAN_GOAL_TOOLSET (set by
/// build_kanban_worker_env; default "restricted" when absent or unrecognized).
async fn filter_for_goal_mode_if_applicable(
    registry: &Arc<RwLock<ToolRegistry>>,
) -> anyhow::Result<()> {
    let worker_mode = std::env::var("HERMES_KANBAN_TASK").is_ok();
    let goal_mode = std::env::var("HERMES_KANBAN_GOAL_MODE")
        .map(|v| v == "1")
        .unwrap_or(false);

    if !worker_mode || !goal_mode {
        return Ok(());
    }

    let preset = std::env::var("HERMES_KANBAN_GOAL_TOOLSET")
        .unwrap_or_else(|_| "restricted".to_string());

    const RESTRICTED: &[&str] = &[
        "kanban_show", "kanban_list", "kanban_comment", "kanban_complete",
        "kanban_block", "kanban_create", "kanban_heartbeat", "kanban_link",
        "kanban_unblock", "kanban_swarm", "kanban_mention", "kanban_decompose",
        "kanban_specify",    // 13 kanban_* tools
        "file_read", "file_write",  // +2 file tools
    ];
    const EXTENDED: &[&str] = &[
        "kanban_show", "kanban_list", "kanban_comment", "kanban_complete",
        "kanban_block", "kanban_create", "kanban_heartbeat", "kanban_link",
        "kanban_unblock", "kanban_swarm", "kanban_mention", "kanban_decompose",
        "kanban_specify",
        "file_read", "file_write",
        "memory_tool", "memory_manager_handle", "skills_tool",  // +3 extended
    ];

    let removed = match preset.as_str() {
        "full" => {
            tracing::info!(target: "kanban.goal_mode", preset = "full", "goal-mode toolset: no filtering");
            0
        }
        "extended" => {
            registry.write().await.retain_by_name(EXTENDED)
        }
        _ => {
            // "restricted" or any unrecognized value → fall back to restricted
            registry.write().await.retain_by_name(RESTRICTED)
        }
    };

    tracing::info!(
        target: "kanban.goal_mode",
        preset = %preset,
        removed = removed,
        "goal-mode toolset filter applied"
    );
    Ok(())
}
```
