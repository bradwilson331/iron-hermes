```rust
// audio_cache.rs — sync sweep (called on startup and from the async loop)
pub fn gc_sweep_audio_cache(dir: &std::path::Path, max_age_days: u32) {
    let cutoff = std::time::SystemTime::now()
        - std::time::Duration::from_secs(max_age_days as u64 * 86400);
    match std::fs::read_dir(dir) {
        Ok(entries) => {
            for entry in entries.flatten() {
                if let Ok(meta) = entry.metadata() {
                    if let Ok(mtime) = meta.modified() {
                        if mtime < cutoff {
                            if let Err(e) = std::fs::remove_file(entry.path()) {
                                // NEVER panic — mirror existing tracing pattern
                                tracing::warn!(
                                    path = %entry.path().display(),
                                    error = %e,
                                    "audio_cache GC: failed to remove file"
                                );
                            }
                        }
                    }
                }
            }
        }
        Err(e) if e.kind() == std::io::ErrorKind::NotFound => {
            // dir not yet created — no-op
        }
        Err(e) => {
            tracing::warn!(error = %e, "audio_cache GC: failed to read dir");
        }
    }
}

pub async fn run_audio_cache_gc_loop(
    dir: std::path::PathBuf,
    max_age_days: u32,
    sweep_interval_secs: u64,
    cancel: tokio_util::sync::CancellationToken,
) {
    use std::time::Duration;
    let interval = Duration::from_secs(sweep_interval_secs);
    loop {
        tokio::select! {
            _ = tokio::time::sleep(interval) => {
                gc_sweep_audio_cache(&dir, max_age_days);
            }
            _ = cancel.cancelled() => {
                tracing::info!("audio_cache GC loop cancelled; exiting");
                break;
            }
        }
    }
}
```

**Don't carry forward:** `kanban_ws.rs` uses `tokio::sync::broadcast` for its tail loop — audio GC has no subscribers, so a plain `CancellationToken` + `sleep` loop is sufficient. Do not adopt the broadcast channel from the kanban analog.

```rust
// crates/iron_hermes_ui/src/server/audio_route.rs
use dioxus::prelude::*;
#[cfg(feature = "server")]
use axum::extract::Path;
#[cfg(feature = "server")]
use axum::response::{IntoResponse, Response};
#[cfg(feature = "server")]
use ironhermes_core::constants::get_hermes_home;

#[get("/audio/:uuid")]
pub async fn serve_audio(
    uuid: String,
) -> impl IntoResponse {
    #[cfg(feature = "server")]
    {
        // Path traversal prevention: parse as UUID first.
        if uuid::Uuid::parse_str(&uuid).is_err() {
            return (axum::http::StatusCode::BAD_REQUEST, "invalid uuid").into_response();
        }
        let audio_cache_dir = get_hermes_home().join("audio_cache");
        let path = audio_cache_dir.join(format!("{uuid}.mp3"));
        // Canonicalize + starts_with guard (Pitfall 4 mitigation).
        let canon = match path.canonicalize() {
            Ok(p) => p,
            Err(_) => return (axum::http::StatusCode::NOT_FOUND, "not found").into_response(),
        };
        if !canon.starts_with(&audio_cache_dir) {
            return (axum::http::StatusCode::FORBIDDEN, "forbidden").into_response();
        }
        match tokio::fs::read(&canon).await {
            Ok(bytes) => (
                [(axum::http::header::CONTENT_TYPE, "audio/mpeg")],
                bytes,
            ).into_response(),
            Err(_) => (axum::http::StatusCode::NOT_FOUND, "not found").into_response(),
        }
    }
    #[cfg(not(feature = "server"))]
    { (axum::http::StatusCode::NOT_FOUND, "server feature required").into_response() }
}
```

**Don't carry forward:** The ws.rs route has no explicit auth gate (operator-local access). Mirror this — no new auth layer. UUID parse validation is sufficient path traversal defense. Do not attempt streaming response for mp3 files (< 1 MB typical, full read is fine per RESEARCH Pattern 4).
