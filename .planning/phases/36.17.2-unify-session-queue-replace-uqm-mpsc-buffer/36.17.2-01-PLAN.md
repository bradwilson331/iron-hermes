```rust
/// Push a multimodal payload onto the per-session sidecar FIFO. Called immediately after a
/// successful SessionQueue::try_push so the payload is FIFO-aligned with the event itself.
pub async fn push_multimodal(
    &self,
    key: &SessionKey,
    payload: (Option<String>, Option<String>),
) {
    let mut map = self.pending_multimodal.lock().await;
    map.entry(key.clone()).or_default().push_back(payload);
}

/// Pop the oldest multimodal payload for this session. Called by the per-chat worker (Plan 02)
/// once per SessionQueue::pop, restoring 1-to-1 FIFO ordering between event and payload.
/// Returns None if no payload is queued (sidecar is empty OR key never had multimodal traffic).
pub async fn take_multimodal(
    &self,
    key: &SessionKey,
) -> Option<(Option<String>, Option<String>)> {
    let mut map = self.pending_multimodal.lock().await;
    map.get_mut(key).and_then(|q| q.pop_front())
}
```

The sidecar uses `tokio::sync::Mutex` (matches `workers`); guards are short-lived and only span O(1) HashMap+VecDeque ops, no awaits inside the critical section.

```rust
    &self,
    event: MessageEvent,
    text_prefix: Option<String>,
    image_data_uri: Option<String>,

    // it cannot cross an await — D-18 + Pitfall 2).
    if let Err(e) = self.session_queue.try_push(&session_key, event) {
```
