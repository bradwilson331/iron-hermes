**Worker Test Driver (helper function):**
The test does NOT boot a full `GatewayRunner` — too heavyweight for unit-test scope. Instead, mirror Plan 02's worker structure inline:

```
async fn spawn_test_worker(
    session_queue: Arc<SessionQueue>,
    uqm: Arc<UserQueueManager>,
    adapter: Arc<dyn PlatformAdapter>,
    session_key: SessionKey,
    handle_log: Arc<Mutex<Vec<String>>>,
    cancel: tokio_util::sync::CancellationToken,
) -> tokio::task::JoinHandle<()> {
    let notify = uqm.notify_for(&session_key).await.expect("notify_for must return Some immediately after WorkerSpawned");
    tokio::spawn(async move {
        loop {
            let event = match session_queue.pop(&session_key) {
                Some(ev) => ev,
                None => {
                    tokio::select! {
                        _ = cancel.cancelled() => break,
                        _ = notify.notified() => continue,
                    }
                }
            };
            if cancel.is_cancelled() { break; }
            // 👁 emission inline before "handle"
            let _ = adapter.add_reaction(&event.chat_id, &event.message_id, "👁").await;
            // Record what handle would have processed (FIFO order check)
            handle_log.lock().unwrap().push(event.message_id.clone());
            if cancel.is_cancelled() { break; }
        }
        uqm.remove(&session_key).await;
    })
}
```

This driver omits `handle_with_multimodal` and the semaphore (both unnecessary for this test's behavioral coverage). It IS the Plan 02 worker structure minus the agent invocation — the same pop-then-react-then-process shape. If Plan 02 deviated from this structure (e.g., 👁 emission moved to a different point), this test driver must match Plan 02's actual structure exactly. **The executor must read Plan 02 SUMMARY first and update this driver to match before writing.**

```
#[tokio::test]
async fn test_5_same_chat_messages_emit_5_eye_reactions_and_fifo_order() {
    let adapter = Arc::new(RecordingPlatformAdapter::new());
    let session_queue = Arc::new(SessionQueue::new());
    let uqm = Arc::new(UserQueueManager::new(adapter.clone() as Arc<dyn PlatformAdapter>, session_queue.clone()));
    let session_key = SessionKey::new(Platform::Telegram, "chat_burst").with_user("user_burst");
    let cancel = tokio_util::sync::CancellationToken::new();
    let handle_log = Arc::new(Mutex::new(Vec::<String>::new()));

    // 1st dispatch — should return WorkerSpawned
    let outcome1 = uqm.dispatch(make_event("chat_burst", "msg_1", "first", "user_burst"), None, None).await.unwrap();
    assert!(matches!(outcome1, DispatchOutcome::WorkerSpawned), "First dispatch must spawn worker, got {:?}", outcome1);

    let worker = spawn_test_worker(
        session_queue.clone(),
        uqm.clone(),
        adapter.clone() as Arc<dyn PlatformAdapter>,
        session_key.clone(),
        handle_log.clone(),
        cancel.clone(),
    ).await;

    // Dispatch 4 more in tight succession — should all return Accepted
    for i in 2..=5 {
        let outcome = uqm.dispatch(
            make_event("chat_burst", &format!("msg_{}", i), &format!("body_{}", i), "user_burst"),
            None,
            None,
        ).await.unwrap();
        assert!(matches!(outcome, DispatchOutcome::Accepted), "Dispatch #{} must return Accepted (worker exists), got {:?}", i, outcome);
    }

    // Allow worker to drain. Poll-with-timeout until handle_log has 5 entries OR 2s elapsed.
    let start = std::time::Instant::now();
    while handle_log.lock().unwrap().len() < 5 {
        if start.elapsed() > std::time::Duration::from_secs(2) {
            panic!("Worker did not drain 5 messages within 2s; got {} so far", handle_log.lock().unwrap().len());
        }
        tokio::time::sleep(std::time::Duration::from_millis(10)).await;
    }

    cancel.cancel();
    let _ = worker.await;

    // Assertion 1: exactly 5 👁 reactions, all targeting chat_burst
    let reactions = adapter.reactions.lock().unwrap();
    let eye_count = reactions.iter().filter(|(c, _, e)| c == "chat_burst" && e == "👁").count();
    assert_eq!(eye_count, 5, "Expected exactly 5 👁 reactions, got {}. Full log: {:?}", eye_count, *reactions);

    // Assertion 2: each msg_1..msg_5 got exactly one 👁 reaction (no dupes)
    for i in 1..=5 {
        let id = format!("msg_{}", i);
        let count = reactions.iter().filter(|(_, m, e)| *m == id && e == "👁").count();
        assert_eq!(count, 1, "msg_{} expected exactly 1 👁, got {}", i, count);
    }
    drop(reactions);

    // Assertion 3: FIFO processing order
    let log = handle_log.lock().unwrap();
    let expected: Vec<String> = (1..=5).map(|i| format!("msg_{}", i)).collect();
    assert_eq!(*log, expected, "Worker processed messages out of FIFO order");

    // Assertion 4: SessionQueue is empty after drain
    assert_eq!(session_queue.len(&session_key), 0, "SessionQueue must be empty after worker drains");
}
```

```
        session_key.clone(), handle_log.clone(), cancel.clone(),

    assert_eq!(session_queue.len(&session_key), 1);
```
