**Step 5 — Test driver `spawn_test_worker` (mirrors Plan 02's worker structure exactly):**

```rust
/// Spawns a single per-chat worker mirroring Plan 02's `runner.rs:958-1024` shape:
/// pop → take_multimodal → 👁 → handle_with_multimodal → loop. The ONLY simplification
/// vs. Plan 02 is omitting the semaphore permit acquisition (acceptable per checker M5
/// — semaphore is bounded-concurrency, not behavioral-coverage relevant).
///
/// The handler call (`handle_with_multimodal`) IS in the loop body so test ordering
/// assertions observe the real handler invocation, not an in-driver shortcut.
async fn spawn_test_worker(
    session_queue: Arc<SessionQueue>,
    uqm: Arc<UserQueueManager>,
    adapter: Arc<dyn PlatformAdapter>,
    handler: Arc<GatewayMessageHandler>,
    session_key: SessionKey,
    cancel: CancellationToken,
) -> tokio::task::JoinHandle<()> {
    let notify = uqm
        .notify_for(&session_key)
        .await
        .expect("notify_for must return Some immediately after WorkerSpawned (Plan 01 invariant)");
    tokio::spawn(async move {
        loop {
            let next_event = match session_queue.pop(&session_key) {
                Some(ev) => ev,
                None => {
                    tokio::select! {
                        _ = cancel.cancelled() => break,
                        _ = notify.notified() => continue,
                    }
                }
            };
            if cancel.is_cancelled() { break; }

            // 👁 emission inline before handle (D-06 step 3, D-08)
            let _ = adapter.add_reaction(&next_event.chat_id, &next_event.message_id, "👁").await;

            // M1 sidecar consumption — one take_multimodal per pop, FIFO lockstep
            let (text_prefix, image_data_uri) = uqm
                .take_multimodal(&session_key)
                .await
                .unwrap_or((None, None));
            let processed = ironhermes_gateway::multimodal::ProcessedAttachments {
                text_prefix,
                image_data_uri,
            };

            // M5 — real handler call path. Under FailingPlatformAdapter, run_agent
            // returns Err at the first placeholder send_message, which the worker logs
            // and ignores. Under RecordingPlatformAdapter, run_agent may proceed further
            // into uncontrolled runtime; tests using Recording adapter MUST inject a
            // FailingPlatformAdapter clone for the handler call only (see Test 1 setup).
            let _ = handler
                .handle_with_multimodal(
                    &next_event,
                    adapter.clone(),
                    cancel.child_token(),
                    processed,
                )
                .await;

            if cancel.is_cancelled() { break; }
        }
        uqm.remove(&session_key).await;
    })
}
```

        Some("PREFIX_C".to_string()),
        Some("data:image/png;base64,IMG_C".to_string()),

    assert!(matches!(r3, DispatchOutcome::Accepted));

    // Wait for worker to process all 3
    let start = Instant::now();
    while adapter.reactions.lock().unwrap().iter().filter(|(c, _, e)| c == "chat_mm" && e == "👁").count() < 3 {
        if start.elapsed() > Duration::from_secs(5) {
            panic!("worker did not process 3 multimodal events within 5s");
        }
        tokio::time::sleep(Duration::from_millis(20)).await;
    }

    cancel.cancel();
    let _ = worker.await;

    // Assertion: sidecar is empty (all 3 payloads consumed in FIFO lockstep with queue pops)
    let leftover = uqm.take_multimodal(&session_key).await;
    assert!(leftover.is_none(), "Sidecar must be empty after worker drains all 3 events, got {:?}", leftover);
