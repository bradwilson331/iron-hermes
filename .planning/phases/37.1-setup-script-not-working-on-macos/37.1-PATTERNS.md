```bash
cmd_uninstall() {
    local purge="${IRONHERMES_PURGE:-false}"
    for arg in "$@"; do
        [ "$arg" = "--purge" ] && purge=true
    done
    # Remove binary
    rm -f "$INSTALL_DIR/ironhermes"
    log_ok "Binary removed from $INSTALL_DIR"
    # Remove PATH lines from shell RCs
    for rc in "$HOME/.bashrc" "$HOME/.zshrc" "$HOME/.profile"; do
        if [ -f "$rc" ]; then
            # Remove the marker comment + PATH line pair
            grep -v "# Added by IronHermes installer" "$rc" | \
            grep -v "export PATH=\"${INSTALL_DIR}:" > "${rc}.tmp" && mv "${rc}.tmp" "$rc"
        fi
    done
    log_ok "PATH entries removed from shell configs"
    # Preserve ~/.ironhermes/ by default; --purge removes it
    if [ "$purge" = "true" ]; then
        rm -rf "$IRONHERMES_HOME"
        log_ok "~/.ironhermes/ purged (--purge flag)"
    else
        log_info "~/.ironhermes/ preserved (use --purge to remove user data)"
    fi
}
```

```rust
/// Writes a commented-default block for `<section>:` if it is absent from `yaml_value`.
/// Pure function — no I/O. Caller does load-mutate-save (matches apply_learning_loop_answer pattern).
pub fn write_defaults_if_absent(yaml_value: &mut serde_yaml::Value, section_key: &str, default_block: &str) {
    if let serde_yaml::Value::Mapping(map) = yaml_value {
        let key = serde_yaml::Value::String(section_key.to_string());
        if !map.contains_key(&key) {
            // Insert default (parsed from default_block YAML string)
            if let Ok(block) = serde_yaml::from_str::<serde_yaml::Value>(default_block) {
                map.insert(key, block);
            }
        }
    }
}
```
