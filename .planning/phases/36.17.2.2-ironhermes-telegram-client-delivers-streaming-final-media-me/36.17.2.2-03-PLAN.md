```rust
// crates/ironhermes-gateway/src/adapter.rs (additions only)

pub use crate::media_tag::{MediaKind, MediaRef, MediaSource};

/// Outbound media-attachment sender. Sibling of `PlatformAdapter`.
///
/// Implementors today: `TelegramAdapter` (plan 04). Discord/Slack/web do NOT
/// implement this trait; their adapters omit it and `GatewayMessageHandler`
/// holds `Option<Arc<dyn MediaSender>>` (D-18) — None on platforms that do
/// not implement it, with the D-19 dispatch loop logging a warn and skipping
/// when None is observed alongside extracted MEDIA tags.
///
/// **D-17 divergence:** No `caption` parameter on any method. The text body
/// is rendered into the placeholder message via `edit_message_markdown_v2`
/// (D-01) BEFORE attachments are sent (D-07), so captions are redundant.
/// This intentionally diverges from hermes-agent §Audio Captions; see
/// CONTEXT.md D-17 for the rationale and the open follow-up.
#[async_trait]
pub trait MediaSender: Send + Sync {
    async fn send_photo(
        &self,
        chat_id: &str,
        source: &MediaSource,
        thread_id: Option<&str>,
    ) -> Result<MessageResponse>;

    async fn send_voice(
        &self,
        chat_id: &str,
        source: &MediaSource,
        thread_id: Option<&str>,
    ) -> Result<MessageResponse>;

    async fn send_audio(
        &self,
        chat_id: &str,
        source: &MediaSource,
        thread_id: Option<&str>,
    ) -> Result<MessageResponse>;

    async fn send_video(
        &self,
        chat_id: &str,
        source: &MediaSource,
        thread_id: Option<&str>,
    ) -> Result<MessageResponse>;

    async fn send_document(
        &self,
        chat_id: &str,
        source: &MediaSource,
        thread_id: Option<&str>,
    ) -> Result<MessageResponse>;

    /// Dispatch based on `MediaKind`. Default impl provided so concrete
    /// adapters need only implement the 5 per-type methods.
    async fn send_media(
        &self,
        chat_id: &str,
        media_ref: &MediaRef,
        thread_id: Option<&str>,
    ) -> Result<MessageResponse> {
        match media_ref.kind {
            MediaKind::Photo    => self.send_photo(chat_id, &media_ref.source, thread_id).await,
            MediaKind::Voice    => self.send_voice(chat_id, &media_ref.source, thread_id).await,
            MediaKind::Audio    => self.send_audio(chat_id, &media_ref.source, thread_id).await,
            MediaKind::Video    => self.send_video(chat_id, &media_ref.source, thread_id).await,
            MediaKind::Document => self.send_document(chat_id, &media_ref.source, thread_id).await,
        }
    }
}
```
