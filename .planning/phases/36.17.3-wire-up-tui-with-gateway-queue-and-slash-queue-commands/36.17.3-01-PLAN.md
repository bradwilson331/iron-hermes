a generic-keyed trait. Per Resolution 5, the trait surface uses `String` and
the gateway impl wraps into `MessageEvent` internally.

Output: New core module, relocated key type, trait impl in gateway, all
existing gateway tests green.

ironhermes_core::queue:
  pub trait MessageQueue<K: Hash + Eq + Clone + Send + Sync + 'static>: Send + Sync {
      fn try_push(&self, key: &K, message: String) -> Result<(), QueueError>;
      fn pop(&self, key: &K) -> Option<String>;
      fn len(&self, key: &K) -> usize;
      fn clear(&self, key: &K);
  }

  #[derive(Debug, thiserror::Error, PartialEq, Eq, Clone)]
  pub enum QueueError {
      #[error("Queue at capacity ({max}) for session {session_key}")]
      CapacityReached { session_key: String, max: usize },
  }
