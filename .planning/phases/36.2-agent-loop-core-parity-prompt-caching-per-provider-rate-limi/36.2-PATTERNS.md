```rust
impl PricingRegistry {
    pub fn lookup(&self, model_id: &str) -> Option<&PricingEntry> {
        // 1. Exact match in cache
        if let Some(e) = self.cache.get(model_id) { return Some(e); }
        // 2. Exact match in static table
        if let Some(e) = self.table.get(model_id) { return Some(e); }
        // 3. Strip provider/ prefix -> try cache, static table
        if let Some((_provider, bare)) = model_id.split_once('/') {
            if let Some(e) = self.cache.get(bare) { return Some(e); }
            if let Some(e) = self.table.get(bare) { return Some(e); }
        }
        None
    }

    /// Lookup with graceful fallback: tracing::warn! + zero-cost entry on miss.
    pub fn lookup_or_zero(&self, model_id: &str) -> PricingEntry {
        match self.lookup(model_id) {
            Some(e) => e.clone(),
            None => {
                tracing::warn!(model_id, "unknown model pricing — cost will be $0");
                PricingEntry { provider: String::new(), input_per_1m_micros: 0,
                    output_per_1m_micros: 0, cache_read_per_1m_micros: 0,
                    cache_creation_per_1m_micros: 0 }
            }
        }
    }

    pub fn merge_cache(&mut self, entries: HashMap<String, PricingEntry>) {
        self.cache.extend(entries);  // cache overrides table (D-06 precedent)
    }
}

impl Default for PricingRegistry {
    fn default() -> Self { Self::new() }
}
```

**Cost calculation** (from RESEARCH.md usage_pricing.py token normalization):
```rust
/// IMPORTANT: Anthropic `input_tokens` in response = tokens AFTER last cache
/// breakpoint only. Total billed input = cache_read + cache_creation + input_tokens.
pub fn compute_cost_micros(entry: &PricingEntry, in_tok: i64, out_tok: i64,
    cache_read: i64, cache_create: i64) -> i64 {
    let input_cost  = (in_tok      * entry.input_per_1m_micros)  / 1_000_000;
    let output_cost = (out_tok     * entry.output_per_1m_micros) / 1_000_000;
    let read_cost   = (cache_read  * entry.cache_read_per_1m_micros) / 1_000_000;
    let create_cost = (cache_create * entry.cache_creation_per_1m_micros) / 1_000_000;
    input_cost + output_cost + read_cost + create_cost
}
```

```rust
use std::time::Duration;

#[derive(Debug, Clone, PartialEq)]
pub enum ProviderError {
    RateLimited { retry_after: Option<Duration> },
    Auth,
    ContextLength,
    Server { status: u16 },
    Transport,
    SchemaInvalid,
    ToolError,
    Unknown,
}

impl ProviderError {
    pub fn should_retry(&self) -> bool {
        matches!(self,
            Self::RateLimited { .. } | Self::Auth | Self::Server { .. } | Self::Transport)
    }
    pub fn should_fallback(&self) -> bool {
        matches!(self,
            Self::RateLimited { .. } | Self::Auth | Self::ContextLength |
            Self::Server { .. } | Self::Transport)
    }
    /// Variant name for storage in usage_events.error_kind column.
    pub fn variant_name(&self) -> &'static str {
        match self {
            Self::RateLimited { .. } => "RateLimited",
            Self::Auth              => "Auth",
            Self::ContextLength     => "ContextLength",
            Self::Server { .. }     => "Server",
            Self::Transport         => "Transport",
            Self::SchemaInvalid     => "SchemaInvalid",
            Self::ToolError         => "ToolError",
            Self::Unknown           => "Unknown",
        }
    }
}

impl From<ProviderError> for (bool, bool) {
    fn from(e: ProviderError) -> Self {
        (e.should_retry(), e.should_fallback())
    }
}

/// New canonical entry point. Internally reuses extract_http_status +
/// is_transport_failure — zero rewrite of those helpers (D-ERR-03).
pub fn classify_llm_error_typed(err: &anyhow::Error) -> ProviderError {
    let err_str = format!("{err:#}");
    let Some(code) = extract_http_status(&err_str) else {
        return if is_transport_failure(&err_str) {
            ProviderError::Transport
        } else {
            ProviderError::Unknown
        };
    };
    match code {
        429              => ProviderError::RateLimited { retry_after: None },
        401 | 403        => ProviderError::Auth,
        400 | 404        => ProviderError::SchemaInvalid,
        413              => ProviderError::ContextLength,
        500 | 502 | 503 | 504 => ProviderError::Server { status: code },
        _                => ProviderError::Unknown,
    }
}
```
