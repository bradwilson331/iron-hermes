```rust
use clap::Subcommand;

#[derive(Subcommand)]
pub enum PricingSubcommand {
    /// Show current pricing table (static table + cache overlay merged).
    List,
    /// Fetch latest pricing from models.dev; write $HERMES_HOME/pricing-cache.json.
    Refresh {
        /// Overwrite cache even if the source returns a partial response.
        /// Without --force, partial responses (zero parsed entries) are rejected.
        #[arg(long)]
        force: bool,
    },
}

pub async fn handle_pricing_command(cmd: PricingSubcommand) -> anyhow::Result<()> {
    match cmd {
        PricingSubcommand::List => cmd_list().await,
        PricingSubcommand::Refresh { force } => cmd_refresh(force).await,
    }
}

async fn cmd_list() -> anyhow::Result<()> {
    use ironhermes_core::{PricingRegistry, PricingCache};
    let mut registry = PricingRegistry::new();
    let cache = PricingCache::load();
    registry.merge_cache(cache.into_pricing_map());
    // Format as a table: model | provider | $in/1M | $out/1M | $cache_read/1M | $cache_create/1M
    // Print to stdout.
    Ok(())
}

async fn cmd_refresh(force: bool) -> anyhow::Result<()> {
    use ironhermes_core::pricing_cache::{fetch_from_models_dev, PricingCache, PricingCacheEntry};
    let fetched = match fetch_from_models_dev().await {
        Ok(entries) => entries,
        Err(e) => {
            eprintln!("hermes pricing refresh: fetch failed — {e}");
            eprintln!("Existing $HERMES_HOME/pricing-cache.json was NOT overwritten.");
            std::process::exit(2);
        }
    };
    if fetched.is_empty() && !force {
        eprintln!("hermes pricing refresh: source returned 0 entries (suspicious schema).");
        eprintln!("Re-run with --force to overwrite cache anyway.");
        std::process::exit(3);
    }
    let cache = PricingCache { entries: fetched };
    cache.save()?;
    println!("hermes pricing refresh: wrote {} entries to {}",
        cache.entries.len(),
        PricingCache::cache_path().display());
    Ok(())
}
```

`fetch_from_models_dev` implementation (replaces the Plan 04 stub in `crates/ironhermes-core/src/pricing_cache.rs`):
```rust
const MODELS_DEV_URL: &str = "https://models.dev/api.json";

pub async fn fetch_from_models_dev() -> anyhow::Result<std::collections::HashMap<String, PricingCacheEntry>> {
    // Mirror the existing models_cache.rs fetch shape — reqwest GET, parse JSON,
    // validate the schema (Tampering threat T-36.2-04-PARSE):
    //   - Response must be valid JSON.
    //   - Top-level shape must contain a "models" or equivalent array/map.
    //   - Each parsed entry must deserialize as PricingEntry (strict i64 fields).
    //   - Unknown fields on PricingEntry are IGNORED (forward-compatible),
    //     but malformed types (string instead of i64) cause parse failure.
    use reqwest::Client;
    use chrono::Utc;

    let client = Client::builder()
        .timeout(std::time::Duration::from_secs(30))
        .build()?;
    let resp = client.get(MODELS_DEV_URL).send().await?;
    if !resp.status().is_success() {
        anyhow::bail!("models.dev returned HTTP {}", resp.status());
    }
    let body: serde_json::Value = resp.json().await?;

    // models.dev API shape (verified 2026-05-25): the top-level is a map of
    // provider -> { models: { model_id -> { cost: { input, output, cache_read, cache_write } } } }.
    // Walk the structure, build PricingEntry per model.
    // (If models.dev changes its API, this parser fails predictably and the
    // cache file is not touched — caller decides whether to --force.)
    let mut out = std::collections::HashMap::new();
    let now = Utc::now();
    if let Some(providers) = body.as_object() {
        for (provider_id, provider_val) in providers {
            if let Some(models) = provider_val.get("models").and_then(|v| v.as_object()) {
                for (model_id, model_val) in models {
                    if let Some(cost) = model_val.get("cost").and_then(|v| v.as_object()) {
                        let entry = PricingEntry {
                            provider: provider_id.clone(),
                            input_per_1m_micros:           cost_field_to_micros(cost.get("input")),
                            output_per_1m_micros:          cost_field_to_micros(cost.get("output")),
                            cache_read_per_1m_micros:      cost_field_to_micros(cost.get("cache_read")),
                            cache_creation_per_1m_micros:  cost_field_to_micros(cost.get("cache_write")),
                        };
                        // Skip entries with all-zero pricing — likely incomplete upstream data.
                        if entry.input_per_1m_micros == 0 && entry.output_per_1m_micros == 0 {
                            continue;
                        }
                        out.insert(model_id.clone(), PricingCacheEntry { pricing: entry, fetched_at: now });
                    }
                }
            }
        }
    }
    Ok(out)
}

/// Convert a JSON numeric field (USD per 1M, possibly float) to i64 micro-USD.
/// Safely handles missing fields (returns 0), float fields, and integer fields.
fn cost_field_to_micros(field: Option<&serde_json::Value>) -> i64 {
    match field {
        Some(v) if v.is_f64() => (v.as_f64().unwrap_or(0.0) * 1_000_000.0) as i64,
        Some(v) if v.is_i64() => v.as_i64().unwrap_or(0) * 1_000_000,
        Some(v) if v.is_u64() => (v.as_u64().unwrap_or(0) as i64) * 1_000_000,
        _ => 0,
    }
}
```

Note: this is the ONE place an f64 conversion crosses the boundary — input from models.dev is typically a float (e.g., 5.0 for $5.00). The conversion to micro-USD happens here and i64 carries from then on (Pitfall 5 contained to the boundary).

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| models.dev HTTPS response → JSON parser → PricingCacheEntry | Untrusted (third-party API); strict typed parsing + len-check defends against spoofed bodies. |
| `--force` CLI flag → cache overwrite | User-supplied; explicit opt-in for partial responses. |
| MODELS_DEV_URL → reqwest::get | Hardcoded compile-time const — adversary cannot redirect via config tampering. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-36.2-09-URL | Spoofing | MODELS_DEV_URL forgery (T-36.2-04-PARSE counterpart) | mitigate | `const MODELS_DEV_URL: &str = "https://models.dev/api.json"` is compile-time. Not surfaced as a config flag, not user-overridable. Adversary cannot redirect `hermes pricing refresh` to a malicious endpoint. |
| T-36.2-09-PARSE | Tampering | Spoofed/malicious models.dev response body | mitigate | (a) `reqwest` validates HTTPS TLS chain by default; (b) JSON parsed with `serde_json::Value::as_object/as_f64/as_i64` — strict types; (c) `cost_field_to_micros` clamps unknown types to 0; (d) without `--force`, empty `HashMap` aborts save with non-zero exit (Test 6); (e) the parsed body never feeds `format!`/`eval`/templating; (f) `PricingEntry` fields are strongly typed i64 — string-injected values fail conversion silently. |
| T-36.2-09-FORCE | Repudiation | `--force` allows silent overwrite of cache | accept | `--force` is an explicit user opt-in. Documented in the `Refresh.force` clap arg help text. Equivalent to the user saying "I know this is incomplete; do it anyway." Exit code on success still 0 — the user remains responsible. |
| T-36.2-09-DOS | Denial of Service | Adversary returns extremely large response → memory exhaustion | accept | `reqwest::Client::timeout(30s)` caps the fetch duration; `resp.json::<serde_json::Value>()` is bounded by reqwest's internal limits. Response sizes from models.dev are typically <100KB. No defensive size cap added; failure mode is process OOM-kill at extreme sizes, which is acceptable for a single-shot CLI command. |
| T-36.2-09-OVERWRITE | Tampering | Failed fetch corrupts existing pricing-cache.json | mitigate | `cmd_refresh` exits BEFORE calling `PricingCache::save()` on any error or empty-without-force path. The existing file is preserved. Verified by Test 6 + Test 8. |
| T-36.2-09-SC | Tampering | npm/pip/cargo installs | mitigate | This plan adds NO new dependencies. Uses existing `reqwest` (from `models_cache.rs`), `serde_json`, `clap`, `chrono`. Package Legitimacy Audit confirms "no new external packages." |
