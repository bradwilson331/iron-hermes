// crates/ironhermes-state/src/lib.rs — new read helper:
#[derive(Debug, Clone, Default)]
pub struct UsageFilter {
    pub session_id: Option<String>,
    pub today_only: bool,
    pub provider: Option<String>,
    pub model: Option<String>,
    pub since_seconds: Option<i64>,    // rolling window relative to now
}

#[derive(Debug, Clone)]
pub struct UsageRollup {
    pub provider: String,
    pub model: String,
    pub in_tok: i64,
    pub out_tok: i64,
    pub cache_read: i64,
    pub cache_create: i64,
    pub cost_usd_micros: i64,
    pub event_count: i64,
}

impl StateStore {
    /// Aggregated rollup by (provider, model) for usage_events matching the filter.
    /// USES `params![]` BINDINGS ONLY — user-supplied filter values never interpolated into SQL.
    pub fn query_usage_events(&self, filter: &UsageFilter) -> Result<Vec<UsageRollup>> {
        let mut sql = String::from(
            "SELECT provider, model, \
             SUM(in_tok), SUM(out_tok), SUM(cache_read), SUM(cache_create), \
             SUM(cost_usd_micros), COUNT(*) \
             FROM usage_events WHERE 1=1"
        );
        let mut params_vec: Vec<Box<dyn rusqlite::ToSql>> = Vec::new();

        if filter.session_id.is_some() {
            sql.push_str(" AND session_id = ?");
            params_vec.push(Box::new(filter.session_id.clone()));
        }
        if filter.today_only {
            // ts is unix epoch ms; midnight today via chrono
            let start_ms = chrono::Local::now()
                .date_naive().and_hms_opt(0, 0, 0).unwrap()
                .and_local_timezone(chrono::Local).unwrap()
                .timestamp_millis();
            sql.push_str(" AND ts >= ?");
            params_vec.push(Box::new(start_ms));
        }
        if let Some(provider) = &filter.provider {
            sql.push_str(" AND provider = ?");
            params_vec.push(Box::new(provider.clone()));
        }
        if let Some(model) = &filter.model {
            sql.push_str(" AND model = ?");
            params_vec.push(Box::new(model.clone()));
        }
        if let Some(secs) = filter.since_seconds {
            let cutoff_ms = chrono::Utc::now().timestamp_millis() - secs * 1000;
            sql.push_str(" AND ts >= ?");
            params_vec.push(Box::new(cutoff_ms));
        }
        sql.push_str(" GROUP BY provider, model ORDER BY SUM(cost_usd_micros) DESC");

        let param_refs: Vec<&dyn rusqlite::ToSql> = params_vec.iter().map(|p| p.as_ref()).collect();
        let mut stmt = self.conn.prepare(&sql)?;
        let rows = stmt.query_map(param_refs.as_slice(), |row| {
            Ok(UsageRollup {
                provider:        row.get(0)?,
                model:           row.get(1)?,
                in_tok:          row.get::<_, Option<i64>>(2)?.unwrap_or(0),
                out_tok:         row.get::<_, Option<i64>>(3)?.unwrap_or(0),
                cache_read:      row.get::<_, Option<i64>>(4)?.unwrap_or(0),
                cache_create:    row.get::<_, Option<i64>>(5)?.unwrap_or(0),
                cost_usd_micros: row.get::<_, Option<i64>>(6)?.unwrap_or(0),
                event_count:     row.get(7)?,
            })
        })?;
        rows.collect()
    }
}

fn cmd_usage(args: &[&str], ctx: &CommandContext) -> CommandResult {
    let store = match &ctx.state_store {
        Some(s) => s.clone(),
        None => return CommandResult::Output("Session storage not configured.".to_string()),
    };

    // Default: current-session totals (no flags).
    let mut filter = UsageFilter {
        session_id: ctx.session_id.clone(),
        ..Default::default()
    };

    let mut iter = args.iter().peekable();
    while let Some(a) = iter.next() {
        match *a {
            "--today" => {
                filter.today_only = true;
                filter.session_id = None;       // --today is across-sessions
            }
            "--provider" => {
                filter.provider = iter.next().map(|s| s.to_string());
                filter.session_id = None;
            }
            "--model" => {
                filter.model = iter.next().map(|s| s.to_string());
                filter.session_id = None;
            }
            "--since" => {
                if let Some(v) = iter.next() {
                    // Parse e.g. "7d", "24h", "30m"
                    if let Some(secs) = parse_since(v) {
                        filter.since_seconds = Some(secs);
                        filter.session_id = None;
                    }
                }
            }
            _ => { /* unknown flag — ignore for now */ }
        }
    }

    let rollups = match store.lock().expect("state store mutex poisoned")
        .query_usage_events(&filter)
    {
        Ok(rows) => rows,
        Err(e) => return CommandResult::Output(format!("Usage query failed: {e}")),
    };

    let table = format_usage_table(&rollups, &filter);
    CommandResult::Output(table)
}

/// Parse "Nd" / "Nh" / "Nm" into seconds. Returns None on invalid input.
fn parse_since(s: &str) -> Option<i64> {
    let s = s.trim();
    if s.is_empty() { return None; }
    let (num_part, unit) = s.split_at(s.len() - 1);
    let n: i64 = num_part.parse().ok()?;
    match unit {
        "d" => Some(n * 86400),
        "h" => Some(n * 3600),
        "m" => Some(n * 60),
        _ => None,
    }
}

fn format_usage_table(rollups: &[UsageRollup], filter: &UsageFilter) -> String {
    if rollups.is_empty() {
        return "No usage data found for this filter.".to_string();
    }
    let mut out = String::new();
    // Header
    out.push_str("Usage\n");
    if let Some(sid) = &filter.session_id {
        out.push_str(&format!("Session: {sid}\n"));
    }
    out.push_str(&format!(
        "{:<10} {:<24} {:>8} {:>8} {:>8} {:>8} {:>10}\n",
        "Provider", "Model", "In tok", "Out tok", "Cache R", "Cache C", "Cost"
    ));
    let mut total_cost: i64 = 0;
    for r in rollups {
        let cost_usd = r.cost_usd_micros as f64 / 1_000_000.0;
        out.push_str(&format!(
            "{:<10} {:<24} {:>8} {:>8} {:>8} {:>8} ${:>9.4}\n",
            r.provider, r.model,
            r.in_tok, r.out_tok, r.cache_read, r.cache_create,
            cost_usd
        ));
        total_cost += r.cost_usd_micros;
    }
    out.push_str(&format!("Total cost: ${:.4}\n", total_cost as f64 / 1_000_000.0));
    out
}
