    pub fn classify_llm_error_typed(err: &anyhow::Error) -> ProviderError {
        let err_str = format!("{err:#}");  // anyhow's chain-formatted string

        // Step 2a: try extract_http_status; on hit, map by code.
        if let Some(code) = crate::agent_loop::extract_http_status(&err_str) {
            return match code {
                429 => ProviderError::RateLimited { retry_after: parse_retry_after(&err_str) },
                401 | 403 => ProviderError::Auth,
                402 => ProviderError::Billing,
                400 => {
                    // 400 has multiple sub-cases — disambiguate by message patterns
                    // (port from Python error_classifier.py: context-overflow,
                    // image-too-large, thinking-sig, llama-grammar, OAuth-beta, generic)
                    classify_400_subcases(&err_str)
                }
                404 => {
                    if err_str.to_lowercase().contains("model") && err_str.to_lowercase().contains("not found") {
                        ProviderError::ModelNotFound
                    } else if err_str.to_lowercase().contains("policy") || err_str.to_lowercase().contains("guardrail") {
                        ProviderError::Auth  // provider_policy_blocked maps to Auth (per RESEARCH.md Python FailoverReason table)
                    } else {
                        ProviderError::SchemaInvalid
                    }
                }
                413 => ProviderError::PayloadTooLarge,
                500 | 502 | 503 | 504 | 529 => ProviderError::Server { status: code },
                _ => ProviderError::Unknown,
            };
        }

        // Step 2b: no status; check transport markers
        if crate::agent_loop::is_transport_failure(&err_str) {
            return ProviderError::Transport;
        }

        // Step 2c: check message-pattern signals without status code (Python tests 25-29):
        // - "rate limit" / "rate_limit" anywhere → RateLimited
        // - "billing" / "credit" / "balance" → Billing
        // - "auth" / "unauthorized" / "invalid api key" → Auth
        let lower = err_str.to_lowercase();
        if lower.contains("rate limit") || lower.contains("rate_limit") || lower.contains("too many requests") {
            return ProviderError::RateLimited { retry_after: None };
        }
        if lower.contains("billing") || lower.contains("credit balance") || lower.contains("insufficient credit") {
            return ProviderError::Billing;
        }
        if lower.contains("unauthorized") || lower.contains("invalid api key") || lower.contains("authentication") {
            return ProviderError::Auth;
        }

        ProviderError::Unknown
    }

    fn parse_retry_after(err_str: &str) -> Option<Duration> {
        // Look for "retry-after: N" or "retry_after: N" (case-insensitive) and parse the integer seconds.
        // Falls back to None if no match.
        // (Reuses no external regex crate — simple substring + parse.)
        let lower = err_str.to_lowercase();
        for needle in &["retry-after:", "retry_after:", "retry after"] {
            if let Some(idx) = lower.find(needle) {
                let tail = &err_str[idx + needle.len()..];
                let secs_str: String = tail.chars().skip_while(|c| !c.is_ascii_digit())
                    .take_while(|c| c.is_ascii_digit()).collect();
                if let Ok(n) = secs_str.parse::<u64>() {
                    return Some(Duration::from_secs(n));
                }
            }
        }
        None
    }

    fn classify_400_subcases(err_str: &str) -> ProviderError {
        let lower = err_str.to_lowercase();
        if lower.contains("context") && (lower.contains("overflow") || lower.contains("too long") || lower.contains("maximum context length")) {
            ProviderError::ContextLength
        } else if lower.contains("image") && (lower.contains("too large") || lower.contains("size limit")) {
            ProviderError::PayloadTooLarge
        } else if lower.contains("thinking") && lower.contains("signature") {
            ProviderError::SchemaInvalid
        } else if lower.contains("json-schema-to-grammar") || lower.contains("llama.cpp") {
            ProviderError::SchemaInvalid
        } else if lower.contains("long context") && lower.contains("beta") {
            ProviderError::SchemaInvalid  // oauth_long_context_beta_forbidden
        } else if lower.contains("multimodal") && lower.contains("tool") {
            ProviderError::ToolError
        } else {
            ProviderError::SchemaInvalid  // format_error
        }
    }
