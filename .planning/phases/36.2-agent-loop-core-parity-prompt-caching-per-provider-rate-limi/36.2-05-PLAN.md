```rust
/// Phase 36.2 (D-CACHE-02): prompt caching configuration.
/// Pre-36.2 configs parse cleanly via #[serde(default)] at the field site.
#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(default)]
pub struct PromptCachingConfig {
    /// Cache TTL for Anthropic cache_control markers. Only "5m" and "1h" are valid.
    pub ttl: CacheTtl,
    /// Enable prompt caching for Anthropic models.
    pub enabled: bool,
}

impl Default for PromptCachingConfig {
    fn default() -> Self { Self { ttl: CacheTtl::OneHour, enabled: true } }
}

#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum CacheTtl {
    #[serde(rename = "5m")]
    FiveMinutes,
    #[serde(rename = "1h")]
    OneHour,
}

impl Default for CacheTtl {
    fn default() -> Self { CacheTtl::OneHour }
}

impl CacheTtl {
    /// String form for the Anthropic cache_control JSON envelope.
    pub fn as_anthropic_ttl(&self) -> &'static str {
        match self {
            CacheTtl::FiveMinutes => "5m",
            CacheTtl::OneHour => "1h",
        }
    }
}
```

cached_layers_must_be_stable() — RESEARCH.md Sub-system A invariant 1:
```rust
/// Phase 36.2 (D-CACHE-04): debug-assert layers 1-8 are free of timestamps / session IDs / per-turn data.
/// Catches silent cache-break bugs at the source.
/// Debug: panic! with the offending substring.
/// Release: tracing::warn! and continue (avoids killing production sessions).
pub(crate) fn cached_layers_must_be_stable(cached_text: &str) {
    // Patterns that indicate per-turn or per-session data leaking into cached layers.
    // ANY occurrence in layers 1-8 invalidates the cache prefix.
    const FORBIDDEN: &[&str] = &[
        // ISO 8601 / RFC 3339 timestamps (any subsecond marker)
        "T2",       // matches "T20" (year 2000+) followed by hours — heuristic
        // UUID-shaped session IDs (8-4-4-4-12 hex digits)
        // Use a coarse heuristic: a substring with 4+ hyphens in a 36-char window
        // (full UUID validation would require regex — skip for performance, rely on
        // session-id specific marker below)
        "session_id=",
        "session_id:",
        "turn_id=",
        "turn_id:",
    ];
    for needle in FORBIDDEN {
        if cached_text.contains(needle) {
            let msg = format!(
                "Phase 36.2 cache stability violation: cached layer contains '{needle}'. \
                 This will break Anthropic prompt caching. Move dynamic content to ephemeral layers 9-10."
            );
            if cfg!(debug_assertions) {
                panic!("{msg}");
            } else {
                tracing::warn!(violation = needle, "{msg}");
            }
        }
    }
}
```
